173:async function openContextFile(path: string, settings: Settings | null): Promise<void> {
174:  const workspaceDir = settings?.workspaceDir;
175:  const isWindows = workspaceDir?.includes("\\") ?? false;
176:  const sep = isWindows ? "\\" : "/";
177:  const abs =
178:    workspaceDir && !/^[a-zA-Z]:[\\/]/.test(path) && !path.startsWith("/")
179:      ? `${workspaceDir.replace(/[\\/]$/, "")}${sep}${path.replace(/^[\\/]+/, "").replace(/\//g, sep)}`
180:      : isWindows
181:        ? path.replace(/\//g, "\\")
182:        : path;
183:  const editor = settings?.editor?.trim();
184:  if (editor) {
185:    await invoke("open_in_editor", { command: editor, path: abs, line: null });
186:    return;
187:  }
188:  await openPath(abs);

191:function buildSessionTree(files: SessionFile[]): TreeNode[] {
192:  const sorted = [...files].sort((a, b) =>
193:    a.path.replace(/\\/g, "/").localeCompare(b.path.replace(/\\/g, "/")),
194:  );
195:  const out: TreeNode[] = [];
196:  const seenDirs = new Set<string>();
197:  for (const f of sorted) {
198:    const displayPath = f.path.replace(/\\/g, "/");
199:    const parts = displayPath.split("/").filter(Boolean);
200:    if (parts.length === 0) continue;
201:    let prefix = "";
202:    for (let i = 0; i < parts.length - 1; i++) {
203:      const seg = parts[i] ?? "";
204:      prefix = prefix ? `${prefix}/${seg}` : seg;
205:      if (!seenDirs.has(prefix)) {
206:        seenDirs.add(prefix);
207:        out.push({ kind: "dir", depth: i, name: seg, key: `d:${prefix}` });
208:      }
209:    }
210:    const leaf = parts[parts.length - 1] ?? "";
211:    out.push({
212:      kind: "file",
213:      depth: parts.length - 1,
214:      name: leaf,
215:      path: displayPath,
216:      key: `f:${f.path}`,
217:      status: f.status,
218:    });
219:  }
220:  return out;
221:}
