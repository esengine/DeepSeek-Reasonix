138:async function openContextFile(path: string, settings: Settings | null): Promise<void> {
139:  const workspaceDir = settings?.workspaceDir;
140:  const isWindows = workspaceDir?.includes("\\") ?? false;
141:  const sep = isWindows ? "\\" : "/";
142:  const abs =
143:    workspaceDir && !/^[a-zA-Z]:[\\/]/.test(path) && !path.startsWith("/")
144:      ? `${workspaceDir.replace(/[\\/]$/, "")}${sep}${path.replace(/^[\\/]+/, "").replace(/\//g, sep)}`
145:      : isWindows
146:        ? path.replace(/\//g, "\\")
147:        : path;
148:  const editor = settings?.editor?.trim();
149:  if (editor) {
150:    await invoke("open_in_editor", { command: editor, path: abs, line: null });
151:    return;
152:  }
153:  await openPath(abs);
154:}

156:function buildSessionTree(files: SessionFile[]): TreeNode[] {
157:  const sorted = [...files].sort((a, b) =>
158:    a.path.replace(/\\/g, "/").localeCompare(b.path.replace(/\\/g, "/")),
159:  );
160:  const out: TreeNode[] = [];
161:  const seenDirs = new Set<string>();
162:  for (const f of sorted) {
163:    const displayPath = f.path.replace(/\\/g, "/");
164:    const parts = displayPath.split("/").filter(Boolean);
165:    if (parts.length === 0) continue;
166:    let prefix = "";
167:    for (let i = 0; i < parts.length - 1; i++) {
168:      const seg = parts[i] ?? "";
169:      prefix = prefix ? `${prefix}/${seg}` : seg;
170:      if (!seenDirs.has(prefix)) {
171:        seenDirs.add(prefix);
172:        out.push({ kind: "dir", depth: i, name: seg, key: `d:${prefix}` });
173:      }
174:    }
175:    const leaf = parts[parts.length - 1] ?? "";
176:    out.push({
177:      kind: "file",
178:      depth: parts.length - 1,
179:      name: leaf,
180:      path: displayPath,
181:      key: `f:${f.path}`,
182:      status: f.status,
183:    });
184:  }
185:  return out;
186:}

281:  const failed = specs.filter((s) => s.status === "failed");
282:  const failedCount = failed.length;
283:  const toolCount = specs.reduce((sum, s) => sum + (s.toolCount ?? s.tools?.length ?? 0), 0);
284:  const filtered =
285:    filter === "ready"
286:      ? specs.filter((s) => s.status === "connected")
287:      : filter === "failed"
288:        ? failed
289:        : [...failed, ...specs.filter((s) => s.status !== "failed")];

291:    if (!onRetry) return;
292:    for (const spec of failed) onRetry(spec.raw);
293:  };
294:  return (
295:    <div className="ctx-block">
296:      <div className="h">
297:        <span>{t("contextPanel.mcpTitle")}</span>
298:        <span className="right">
299:          {specs.length === 0
300:            ? "—"
301:            : bridged
302:              ? t("contextPanel.mcpReadyAll", { count: specs.length })

305:      </div>
306:      {specs.length === 0 ? (
307:        <div className="ctx-empty">{t("contextPanel.mcpEmpty")}</div>

316:            <span>{t("contextPanel.mcpHealthTools", { count: toolCount })}</span>

324:                data-active={filter === kind}

326:              >
327:                {kind === "all"
328:                  ? t("contextPanel.mcpFilterAll")
329:                  : kind === "ready"
330:                    ? t("contextPanel.mcpFilterReady")
